#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define N 8
#define LEN 20

typedef struct Node {
    char key[LEN];
    struct Node *left, *right;
} Node;

int bstComparisons = 0;

Node *newNode(const char *key) {
    Node *node = (Node *)malloc(sizeof(Node));
    if (node == NULL) {
        printf("Memory allocation failed.\n");
        return NULL;
    }
    strcpy(node->key, key);
    node->left = node->right = NULL;
    return node;
}

Node *insert(Node *root, const char *key) {
    if (root == NULL) return newNode(key);

    int cmp = strcmp(key, root->key);
    if (cmp < 0)
        root->left = insert(root->left, key);
    else if (cmp > 0)
        root->right = insert(root->right, key);
    return root;
}

void inorder(Node *root) {
    if (root != NULL) {
        inorder(root->left);
        printf("%s ", root->key);
        inorder(root->right);
    }
}

int height(Node *root) {
    if (root == NULL) return 0;
    int leftHeight = height(root->left);
    int rightHeight = height(root->right);
    return 1 + (leftHeight > rightHeight ? leftHeight : rightHeight);
}

int bstSearch(Node *root, const char *key) {
    while (root != NULL) {
        bstComparisons++;
        int cmp = strcmp(key, root->key);
        if (cmp == 0) return 1;
        root = (cmp < 0) ? root->left : root->right;
    }
    return 0;
}

int linearSearch(char keys[][LEN], int n, const char *key, int *comparisons) {
    *comparisons = 0;
    for (int i = 0; i < n; i++) {
        (*comparisons)++;
        if (strcmp(keys[i], key) == 0) return i;
    }
    return -1;
}

int main(void) {
    char keys[N][LEN] = {
        "A102", "A25", "A7", "B100",
        "B12", "A120", "B3", "A45"
    };

    Node *root = NULL;
    for (int i = 0; i < N; i++) {
        root = insert(root, keys[i]);
        if (root == NULL) {
            printf("Could not build the tree.\n");
            return 1;
        }
    }

    printf("Identification numbers inserted in given order:\n");
    for (int i = 0; i < N; i++) printf("%s ", keys[i]);

    printf("\n\nInorder traversal:\n");
    inorder(root);

    printf("\n\nBST height (levels): %d\n", height(root));

    const char *queries[] = {"A7", "B12", "A45", "C9"};
    int queryCount = (int)(sizeof(queries) / sizeof(queries[0]));

    printf("\nSearch comparison (each strcmp counts as one key comparison):\n");
    printf("Key\tBST result\tBST comparisons\tLinear result\tLinear comparisons\n");
    for (int i = 0; i < queryCount; i++) {
        bstComparisons = 0;
        int foundBST = bstSearch(root, queries[i]);
        int bstCount = bstComparisons;

        int linearCount = 0;
        int index = linearSearch(keys, N, queries[i], &linearCount);

        printf("%s\t%s\t\t%d\t\t%s\t\t%d\n",
               queries[i],
               foundBST ? "Found" : "Not found",
               bstCount,
               index >= 0 ? "Found" : "Not found",
               linearCount);
    }

    return 0;
}
